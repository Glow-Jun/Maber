```csharp
using System.Text;

namespace Maber.Core;

public readonly record struct DecodeResult(string Text, string EncodingName, bool HasBom);

public static class EncodingDetector
{
    static EncodingDetector()
    {
        Encoding.RegisterProvider(CodePagesEncodingProvider.Instance);
    }

    private static readonly Encoding StrictUtf8 =
        new UTF8Encoding(encoderShouldEmitUTF8Identifier: false,
                         throwOnInvalidBytes: true);

    public static DecodeResult Decode(byte[] bytes)
    {
        if (bytes.Length == 0)
            return new DecodeResult(string.Empty, "utf-8", false);

        if (bytes.Length >= 3 && bytes[0] == 0xEF && bytes[1] == 0xBB && bytes[2] == 0xBF)
            return new DecodeResult(Encoding.UTF8.GetString(bytes, 3, bytes.Length - 3), "utf-8", true);

        try
        {
            return new DecodeResult(StrictUtf8.GetString(bytes), "utf-8", false);
        }
        catch (DecoderFallbackException)
        {
            return new DecodeResult(Encoding.GetEncoding("GBK").GetString(bytes), "gbk", false);
        }
    }

    public static byte[] Encode(string text, string encodingName, bool hasBom)
    {
        var enc = encodingName == "gbk"
            ? Encoding.GetEncoding("GBK")
            : new UTF8Encoding(encoderShouldEmitUTF8Identifier: false);
        var body = enc.GetBytes(text);
        if (!hasBom) return body;
        var preamble = enc is UTF8Encoding ? [0xEF, 0xBB, 0xBF] : enc.GetPreamble();
        return [.. preamble, .. body];
    }
}
```

注意：GBK 的 GetPreamble 为空，Encode 时 hasBom 且 gbk 的组合实际不会出现（GBK 无 BOM），preamble 逻辑只为 utf-8 生效。

```csharp
namespace Maber.Core;

public readonly record struct LoadedDocument(
    string Path, string Text, string EncodingName, bool HasBom);

public sealed class FileDocumentService
{
    /// <summary>以 FileShare.ReadWrite 打开：我们读的同时，别的程序可以写。</summary>
    public async Task<LoadedDocument> LoadAsync(string path, CancellationToken ct = default)
    {
        await using var fs = new FileStream(
            path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite,
            bufferSize: 64 * 1024, useAsync: true);
        using var ms = new MemoryStream();
        await fs.CopyToAsync(ms, ct);
        var (text, enc, bom) = EncodingDetector.Decode(ms.ToArray());
        return new LoadedDocument(path, text, enc, bom);
    }

    /// <summary>写临时文件后原子替换，避免写入中断导致文件损坏。</summary>
    public async Task SaveAsync(string path, string text, string encodingName, bool hasBom)
    {
        var bytes = EncodingDetector.Encode(text, encodingName, hasBom);
        var dir = Path.GetDirectoryName(Path.GetFullPath(path))!;
        var tmp = Path.Combine(dir, $".{Path.GetFileName(path)}.{Guid.NewGuid():N}.tmp");

        await using (var fs = new FileStream(
            tmp, FileMode.Create, FileAccess.Write, FileShare.None,
            bufferSize: 64 * 1024, useAsync: true))
        {
            await fs.WriteAsync(bytes, ct: default);
            await fs.FlushAsync();
        }

        try
        {
            if (File.Exists(path)) File.Replace(tmp, path, destinationBackupPath: null);
            else File.Move(tmp, path);
        }
        catch (IOException)
        {
            File.Move(tmp, path, overwrite: true);
        }
    }
}
```

```csharp
using System.IO;

namespace Maber.Core;

/// <summary>
/// 监视单个文件的外部修改。FSW 为主、按 LastWriteTimeUtc+Length 轮询兜底
/// （网盘/网络路径上 FSW 不可靠）。自己的写入由调用方比对内容后忽略。
/// </summary>
public sealed class ExternalFileWatcher : IDisposable
{
    public int PollIntervalMs { get; init; } = 2000;

    private FileSystemWatcher? _fsw;
    private Timer? _poll;
    private string? _path;
    private (DateTime Utc, long Len) _last;
    private bool _suspended;
    private readonly SemaphoreSlim _gate = new(1, 1);

    public event EventHandler? ExternalChanged;
    public event EventHandler? ExternalDeleted;

    public void Attach(string path)
    {
        Detach();
        _path = Path.GetFullPath(path);
        _last = Stat(_path);

        var dir = Path.GetDirectoryName(_path)!;
        var name = Path.GetFileName(_path);
        _fsw = new FileSystemWatcher(dir, name)
        {
            NotifyFilter = NotifyFilters.LastWrite | NotifyFilters.Size | NotifyFilters.FileName,
        };
        _fsw.Changed += OnFswEvent;
        _fsw.Renamed += OnFswEvent;
        _fsw.EnableRaisingEvents = true;

        _poll = new Timer(_ => _ = PollAsync(), null, PollIntervalMs, PollIntervalMs);
    }

    /// <summary>自己保存前调用，避免把自写当外部修改。</summary>
    public void SuspendOnce()
    {
        _suspended = true;
        if (_path is not null) _last = Stat(_path);
    }

    public void Detach()
    {
        _fsw?.Dispose(); _fsw = null;
        _poll?.Dispose(); _poll = null;
        _path = null;
    }

    private void OnFswEvent(object s, EventArgs e) => _ = PollAsync();

    private async Task PollAsync()
    {
        if (_path is null || _suspended) return;
        await _gate.WaitAsync();
        try
        {
            if (_path is null || _suspended) return;
            var now = Stat(_path);
            if (now == _last) return;

            if (now.Utc == default)
            {
                _last = now;
                ExternalDeleted?.Invoke(this, EventArgs.Empty);
                return;
            }

            _last = now;
            ExternalChanged?.Invoke(this, EventArgs.Empty);
        }
        finally
        {
            // 短暂忽略 FSW 的连发（编辑器常触发多次 Changed）
            await Task.Delay(100);
            if (_path is not null) _last = Stat(_path);
            _suspended = false;
            _gate.Release();
        }
    }

    private static (DateTime Utc, long Len) Stat(string path)
        => File.Exists(path) ? (File.GetLastWriteTimeUtc(path), new FileInfo(path).Length) : (default, -1);

    public void Dispose() => Detach();
}
```

```ts
import MarkdownIt from "markdown-it";

export function buildRenderer(): MarkdownIt {
  const md = new MarkdownIt({ html: true, linkify: true, breaks: false })
    .enable(["table", "strikethrough"]);

  // 标题带源行号（heading_open token.map[0] 即行首）
  const defaultHeadingOpen = md.renderer.rules.heading_open ?? ((t, i, e, _o, s) => s.renderToken(t, i, e));
  md.renderer.rules.heading_open = (tokens, idx, opts, env, self) => {
    const map = tokens[idx].map;
    if (map) tokens[idx].attrSet("data-line", String(map[0]));
    return defaultHeadingOpen(tokens, idx, opts, env, self);
  };

  // 任务列表：list_item_open 检测首行 `- [ ]`/`- [ ]`，包 checkbox
  md.core.ruler.push("maber_task_items", (state) => {
    for (let i = 0; i < state.tokens.length; i++) {
      const tok = state.tokens[i];
      if (tok.type !== "list_item_open" || !tok.map) continue;
      const lineStart = tok.map[0];
      const line = state.src.slice(state.bMarks[lineStart], state.eMarks[lineStart]);
      const m = /^(\s*[-*+]\s+)\[( |x|X)\](\s+)/.exec(line);
      if (!m) continue;
      tok.attrJoin("class", "task");
      tok.attrSet("data-line", String(lineStart));
      tok.attrSet("data-checked", m[2].toLowerCase() === "x" ? "true" : "false");
    }
  });

  // frontmatter：文首 --- 块抽出进 env，不参与渲染
  md.core.ruler.push("maber_frontmatter", (state) => {
    if (state.tokens.length === 0) return;
    const first = state.tokens[0];
    if (first.type !== "fence" || first.info !== "" || first.tag !== "code") return;
    const src = state.src;
    if (!src.startsWith("---")) return;
    const end = src.indexOf("\n---", 3);
    if (end === -1) return;
    const block = src.slice(4, end);
    (state.env as Record<string, unknown>).frontmatter = parseYamlFlat(block);
    // 从 token 流里删掉这个 fence 及其后空段
    state.tokens.splice(0, 1);
  });

  return md;
}

/** v0.1 只做一层 `key: value` 解析，够用为先 */
export function parseYamlFlat(block: string): Record<string, string> {
  const out: Record<string, string> = {};
  for (const raw of block.split("\n")) {
    const line = raw.trim();
    if (!line) continue;
    const idx = line.indexOf(":");
    if (idx <= 0) continue;
    out[line.slice(0, idx).trim()] = line.slice(idx + 1).trim().replace(/^["']|["']$/g, "");
  }
  return out;
}
```

```ts
import type MarkdownIt from "markdown-it";
import type StateInline from "markdown-it/lib/rules_inline/state_inline.mjs";

/** 依序注册：高亮、注释、标签、wikilink/embed、callout */
export function obsidianPlugins(md: MarkdownIt): MarkdownIt {
  inlineRule(md, "highlight", "==", (s) => `<mark>${s}</mark>`);
  inlineRule(md, "maber_comment", "%%", () => "");
  addTagRule(md);
  addWikilinkRule(md);
  addCalloutRule(md);
  return md;
}

/** 通用成对标记规则：==xx==、%%xx%% */
function inlineRule(md: MarkdownIt, name: string, marker: string, wrap: (s: string) => string) {
  md.inline.ruler.before("emphasis", name, (state: StateInline, silent) => {
    const start = state.pos;
    if (state.src.slice(start, start + marker.length) !== marker) return false;
    const end = state.src.indexOf(marker, start + marker.length);
    if (end === -1) return false;
    const content = state.src.slice(start + marker.length, end);
    if (!content.trim()) return false;
    if (!silent) {
      const tok = state.push("html_inline", "", 0);
      tok.content = wrap(md.utils.escapeHtml(content));
    }
    state.pos = end + marker.length;
    return true;
  });
}

function addTagRule(md: MarkdownIt) {
  md.inline.ruler.before("link", "maber_tag", (state, silent) => {
    const start = state.pos;
    if (state.src[start] !== "#") return false;
    const prev = start === 0 ? " " : state.src[start - 1];
    if (!/[\s(（>]/.test(prev)) return false;                 // 必须在空白/括号后，避免命中标题和 URL 锚点
    const m = /^#([\p{L}\p{N}][\p{L}\p{N}_/-]*)/u.exec(state.src.slice(start));
    if (!m) return false;
    if (!silent) {
      const tok = state.push("html_inline", "", 0);
      tok.content = `<span class="tag">#${md.utils.escapeHtml(m[1])}</span>`;
    }
    state.pos += m[0].length;
    return true;
  });
}

function addWikilinkRule(md: MarkdownIt) {
  md.inline.ruler.before("link", "maber_wikilink", (state, silent) => {
    const src = state.src;
    const start = state.pos;
    const isEmbed = src.slice(start, start + 3) === "![[";

    const open = isEmbed ? start + 3 : start + 2;
    if (src.slice(isEmbed ? start + 1 : start, (isEmbed ? start + 1 : start) + 2) !== "[[") return false;
    const close = src.indexOf("]]", open);
    if (close === -1) return false;
    const raw = src.slice(open, close).replace(/\\/, "");
    const [targetRaw, alias] = raw.split("|");
    const target = targetRaw.split("#")[0].trim();
    if (!target && !alias) return false;

    if (!silent) {
      const tok = state.push("html_inline", "", 0);
      if (isEmbed && /\.(png|jpe?g|gif|webp|svg|bmp|avif)$/i.test(target)) {
        tok.content = `<img class="embed" src="file://${encodeURIComponent(target)}" alt="${md.utils.escapeHtml(target)}">`;
      } else if (isEmbed) {
        tok.content = `<span class="embed-placeholder">![[]] ${md.utils.escapeHtml(target)}</span>`;
      } else {
        const label = (alias ?? target).split("#")[0] || target;
        tok.content = `<a class="wikilink" data-target="${md.utils.escapeHtml(target)}" href="#">${md.utils.escapeHtml(label)}</a>`;
      }
    }
    state.pos = close + 2;
    return true;
  });
}

function addCalloutRule(md: MarkdownIt) {
  const defaultBq = md.renderer.rules.blockquote_open
    ?? ((tokens, idx, opts, env, self) => self.renderToken(tokens, idx, opts));
  md.renderer.rules.blockquote_open = (tokens, idx, opts, env, self) => {
    // 找 blockquote 内第一个 inline token，识别 `> [!type]- title`
    const inner = tokens[idx];
    const next = tokens[idx + 1];
    const inline = next?.type === "inline" ? next : undefined;
    const first = inline?.content ?? "";
    const m = /^\[!(\w+)\](-?)\s*(.*)$/.exec(first.trimStart());
    if (!m) return defaultBq(tokens, idx, opts, env, self);

    const [, kind, fold, title] = m;
    inline!.content = title || kind.toUpperCase();
    inner.attrJoin("class", `callout callout-${kind.toLowerCase()}`);
    if (fold === "-") inner.attrSet("data-collapsed", "true");
    return `<div ${self.renderAttrs(inner)}><div class="callout-title">${md.utils.escapeHtml(inline!.content)}</div><div class="callout-body">`;
  };
  const defaultBqClose = md.renderer.rules.blockquote_close
    ?? ((tokens, idx, opts, env, self) => self.renderToken(tokens, idx, opts));
  md.renderer.rules.blockquote_close = (tokens, idx, opts, env, self) => {
    const open = tokens[idx - 0]; // close 配对不携带 class，检查栈深即可：简化为固定尾部
    return `</div></div>` + defaultBqClose(tokens, idx, opts, env, self).replace("<blockquote>", "");
  };
}
```

注意：callout 的 close 规则里默认 `<blockquote>` 开标签已被改写为 `<div>`，close 侧要输出 `</div>` 而非 `</blockquote>`——实现时以最终 HTML 正确为准，测试即裁判；如该实现细节在实际调试中过不去，允许改为更稳的 core 规则方案（把 blockquote token 流整体改写为 callout 专用 token），但对外行为（测试断言）不变。

```ts
import type MarkdownIt from "markdown-it";
import katex from "katex";

export function mathPlugin(md: MarkdownIt): MarkdownIt {
  const render = (tex: string, display: boolean) => {
    try {
      return katex.renderToString(tex, { displayMode: display, throwOnError: true, output: "html" });
    } catch {
      return `<span class="katex-error">${md.utils.escapeHtml(tex)}</span>`;
    }
  };

  // ---- inline $...$ ----
  md.inline.ruler.before("escape", "maber_math_inline", (state, silent) => {
    const start = state.pos;
    if (state.src[start] !== "$") return false;
    if (state.src[start - 1] === "\\") return false;
    // $ 后不允许直接空白（排除 "价格 $ 5"），$ 前一位若是数字则排除（排除 100$ 买）
    const after = state.src[start + 1];
    if (after === undefined || /\s/.test(after)) return false;
    const before = start === 0 ? "" : state.src[start - 1];
    if (/\d/.test(before) && !silent) { /* fallthrough：允许 100$ 的场景由闭合计价 */ }

    let end = start + 1;
    let found = -1;
    while ((end = state.src.indexOf("$", end)) !== -1) {
      if (state.src[end - 1] === "\\") { end += 1; continue; }
      if (end === start + 1) { break; }               // $$ 由块规则处理
      if (/\s/.test(state.src[end - 1])) { break; }   // 闭合前是空白 → 不是公式
      found = end;
      break;
    }
    if (found === -1) return false;

    if (!silent) {
      const tex = state.src.slice(start + 1, found).replace(/\\\$/g, "$");
      const tok = state.push("html_inline", "", 0);
      tok.content = render(tex, false);
    }
    state.pos = found + 1;
    return true;
  });

  // ---- block $$...$$ ----
  md.block.ruler.before("fence", "maber_math_block", (state, startLine, endLine, silent) => {
    const startPos = state.bMarks[startLine] + state.tShift[startLine];
    if (state.src.slice(startPos, startPos + 2) !== "$$") return false;

    let line = startLine;
    let content = "";
    const firstLine = state.src.slice(startPos + 2, state.eMarks[startLine]).trim();
    if (firstLine.endsWith("$$") && firstLine.length >= 2) {
      content = firstLine.slice(0, -2);
      line = startLine;
    } else {
      content = firstLine ? firstLine + "\n" : "";
      let closed = false;
      for (line = startLine + 1; line < endLine; line++) {
        const ls = state.bMarks[line] + state.tShift[line];
        const l = state.src.slice(ls, state.eMarks[line]);
        if (l.trimEnd() === "$$") { closed = true; break; }
        content += l + "\n";
      }
      if (!closed) return false;
    }
    if (silent) return true;

    const tok = state.push("html_block", "", 0);
    tok.content = render(content, true);
    state.line = line + 1;
    return true;
  });

  return md;
}
```

```ts
let hits: HTMLElement[] = [];
let current = -1;

export function find(root: HTMLElement, query: string, direction: 1 | -1): number {
  if (!query) return -1;
  if (current === -1 || hits[0]?.dataset.query !== query) buildHits(root, query);
  if (hits.length === 0) return 0;
  current = (current + direction + hits.length) % hits.length;
  hits.forEach((h, i) => h.classList.toggle("current", i === current));
  hits[current].scrollIntoView({ block: "center", behavior: "smooth" });
  return hits.length;
}

export function clearFind(root: HTMLElement) {
  root.querySelectorAll("mark.find-hit").forEach((m) => {
    const parent = m.parentNode!;
    parent.replaceChild(document.createTextNode(m.textContent ?? ""), m);
    parent.normalize();
  });
  hits = []; current = -1;
}

function buildHits(root: HTMLElement, query: string) {
  clearFind(root);
  const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
  const targets: Text[] = [];
  let node: Node | null;
  while ((node = walker.nextNode())) {
    if ((node as Text).data.toLowerCase().includes(query.toLowerCase())) targets.push(node as Text);
  }
  for (const t of targets) {
    const frag = document.createDocumentFragment();
    let rest = t.data;
    let idx: number;
    while ((idx = rest.toLowerCase().indexOf(query.toLowerCase())) !== -1) {
      if (idx > 0) frag.appendChild(document.createTextNode(rest.slice(0, idx)));
      const mark = document.createElement("mark");
      mark.className = "find-hit";
      mark.dataset.query = query;
      mark.textContent = rest.slice(idx, idx + query.length);
      frag.appendChild(mark);
      hits.push(mark);
      rest = rest.slice(idx + query.length);
    }
    if (rest) frag.appendChild(document.createTextNode(rest));
    t.parentNode!.replaceChild(frag, t);
  }
}
```

```ts
  if (next === "source") {
    editor.show(currentText);
    content.style.display = "none";
  } else {
    const text = editor.hide();
    content.style.display = "";
    if (text) renderDocument(text);
  }
  document.body.dataset.mode = mode;
}

// 任务复选框点击 → 通知宿主写回源文件
content.addEventListener("click", (e) => {
  const t = e.target as HTMLElement;
  if (t.classList.contains("task-marker")) {
    const item = t.closest('[data-line][data-checked]')!;
    const checked = item.getAttribute("data-checked") !== "true";
    item.setAttribute("data-checked", String(checked));
    t.classList.toggle("checked", checked);
    send({ type: "checkboxToggled", line: Number(item.getAttribute("data-line")), checked });
  }
  const link = t.closest("a");
  if (link) {
    e.preventDefault();
    if (link.classList.contains("wikilink")) return; // v0.1 不跳转
    send({ type: "openLink", href: link.getAttribute("href") ?? "" });
  }
});

onHostMessage(async (msg) => {
  switch (msg.type) {
    case "loadDocument":
      if (editor.visible) { currentText = msg.content; editor.show(msg.content); }
      else renderDocument(msg.content);
      break;
    case "setMode": setMode(msg.mode); break;
    case "setTheme":
      document.body.dataset.theme = msg.theme;
      setMermaidTheme(msg.theme);
      break;
    case "scrollToHeading": {
      const el = content.querySelector(`[data-line="${msg.line}"]`);
      el?.scrollIntoView({ block: "start", behavior: "smooth" });
      break;
    }
    case "find": {
      const target: HTMLElement = editor.visible
        ? document.querySelector("#editor-host")!
        : content;
      if (editor.visible) { /* v0.1 源码模式查找复用 CM 面板，快捷键提示交给宿主 */ break; }
      find(target, msg.query, msg.direction);
      break;
    }
  }
});

// 给宿主的取源入口
(window as any).__maber = { getSource: () => editor.visible ? editor.hide.toString() && currentText : currentText };
send({ type: "ready" });
```

- [ ] **Step 6: MainWindow.xaml.cs（核心宿主逻辑，全部写在一个类里，v0.1 不拆）**

```csharp
using System.Text;
using System.Text.Json;
using Microsoft.UI;
using Microsoft.UI.Xaml;
using Microsoft.UI.Xaml.Controls;
using Microsoft.UI.Xaml.Input;
using Maber.Core;
using Windows.System;

namespace Maber.App;

public sealed partial class MainWindow : Window
{
    private readonly FileDocumentService _files = new();
    private readonly ExternalFileWatcher _watcher = new();
    private readonly Dictionary<string, JsonElement> _msg = new();

    private LoadedDocument? _doc;          // 当前文件（含编码信息）
    private string _currentText = "";      // 壳侧权威源文本
    private bool _dirty;                   // 相对磁盘的内容
    private bool _ready;                   // web 内核就绪
    private bool _sourceMode;
    private List<(int level, string text, int line)> _outline = new();
    private static readonly JsonSerializerOptions JsonOpts = new(JsonSerializerDefaults.Web);

    public MainWindow()
    {
        InitializeComponent();
        Title = "Maber";
        ExtendsContentIntoTitleBar = true;
        SetTitleBar(AppTitleBar);

        Web.WebMessageReceived += OnWebMessage;
        Web.NavigationCompleted += async (_, _) => { _ready = true; await SendThemeAsync(); };
        _watcher.ExternalChanged += (_, _) => DispatcherQueue.TryEnqueue(OnExternalChanged);
        _watcher.ExternalDeleted += (_, _) => DispatcherQueue.TryEnqueue(
            () => ShowBar("文件已被外部删除", allowReload: false));
        Activated += async (_, e) => { if (e.WindowActivationState != WindowActivationState.Deactivated) await SendThemeAsync(); };

        var keyArgs = Environment.GetCommandLineArgs().Skip(1);
        OpenFileCommand = new XamlUICommand();
        _ = InitializeAsync(keyArgs.FirstOrDefault(p => File.Exists(p)));
    }

    private async Task InitializeAsync(string? path)
    {
        await Web.EnsureCoreWebView2Async();
        Web.CoreWebView2.SetVirtualHostNameToFolderMapping(
            "app.local", Path.Combine(AppContext.BaseDirectory, "wwwroot"),
            Microsoft.Web.WebView2.Core.CoreWebView2HostResourceAccessKind.Allow);
        if (path is null) { EmptyState.IsOpen = true; return; }
        await LoadFileAsync(path);
    }

    // ---------- 文件 ----------
    private async Task LoadFileAsync(string path)
    {
        _doc = await _files.LoadAsync(path);
        _currentText = _doc.Text;
        _dirty = false;
        _watcher.Attach(path);
        TitleText.Text = Path.GetFileName(path);
        Post(new { type = BridgeMessages.LoadDocument, name = Path.GetFileName(path), content = _currentText });
    }

    private async Task SaveAsync()
    {
        if (_doc is null) return;
        _watcher.SuspendOnce();
        await _files.SaveAsync(_doc.Path, _currentText, _doc.EncodingName, _doc.HasBom);
        _dirty = false;
    }

    private void OnExternalChanged()
    {
        if (_doc is null) return;
        // 竞态兜底：读最新内容；与我们手里的权威文本一致则是自写回声
        var fresh = _files.LoadAsync(_doc.Path).GetAwaiter().GetResult();
        if (fresh.Text == _currentText) return;
        if (!_dirty) { _doc = fresh; _currentText = fresh.Text; Post(new { type = BridgeMessages.LoadDocument, name = Path.GetFileName(_doc.Path), content = _currentText }); }
        else { ExternalBar.Title = "文件已被外部修改，你有未保存的更改"; ExternalBar.IsOpen = true; }
    }

    private async void OnReloadExternal(object s, RoutedEventArgs e)
    {
        ExternalBar.IsOpen = false;
        if (_doc is null) return;
        var fresh = await _files.LoadAsync(_doc.Path);
        _doc = fresh; _dirty = false;
        _currentText = fresh.Text;
        Post(new { type = BridgeMessages.LoadDocument, name = Path.GetFileName(_doc.Path), content = _currentText });
    }

    private void ShowBar(string msg, bool allowReload) { ExternalBar.Title = msg; ExternalBar.IsOpen = true; }

    // ---------- 桥接 ----------
    private void OnWebMessage(CoreWebView2.Disp? _, object? __) { } // 占位防误写，真实签名见下

    private void OnWebMessage(object sender, Microsoft.Web.WebView2.Core.CoreWebView2WebMessageReceivedEventArgs e)
    {
        var json = JsonDocument.Parse(e.WebMessageAsJson);
        var type = json.RootElement.GetProperty("type").GetString();
        switch (type)
        {
            case BridgeMessages.CheckboxToggled:
                ToggleTaskLine(json.RootElement.GetProperty("line").GetInt32(), json.RootElement.GetProperty("checked").GetBoolean());
                _ = SaveAsync();
                break;
            case BridgeMessages.ContentChanged:
                _currentText = json.RootElement.GetProperty("content").GetString() ?? _currentText;
                _dirty = true;
                break;
            case BridgeMessages.OutlineChanged:
                _outline = json.RootElement.GetProperty("items").EnumerateArray()
                    .Select(i => (i.GetProperty("level").GetInt32(), i.GetProperty("text").GetString() ?? "", i.GetProperty("line").GetInt32()))
                    .ToList();
                break;
            case BridgeMessages.OpenLink:
                var href = json.RootElement.GetProperty("href").GetString() ?? "";
                if (href.StartsWith("http")) _ = Launcher.LaunchUriAsync(new Uri(href));
                break;
        }
    }

    private void Post(object msg) => Web.CoreWebView2.PostWebMessageAsJson(JsonSerializer.Serialize(msg, JsonOpts));

    private void ToggleTaskLine(int line, bool checkedNow)
    {
        var lines = new List<string>(_currentText.Split('\n'));
        if (line < 0 || line >= lines.Count) return;
        lines[line] = checkedNow
            ? System.Text.RegularExpressions.Regex.Replace(lines[line], r"\[( |x|X)\]", "[x]", 1)
            : System.Text.RegularExpressions.Regex.Replace(lines[line], r"\[( |x|X)\]", "[ ]", 1);
        _currentText = string.Join("\n", lines);
        _dirty = true;
    }

    // ---------- 主题跟随系统 ----------
    private async Task SendThemeAsync()
    {
        if (!_ready) return;
        var light = Microsoft.Win32.Registry.GetValue(
            @"HKEY_CURRENT_USER\Software\Microsoft\Windows\CurrentVersion\Themes\Personalize",
            "AppsUseLightTheme", 1) is int v && v == 1;
        Post(new { type = BridgeMessages.SetTheme, theme = light ? "light" : "dark" });
        await Task.CompletedTask;
    }

    // ---------- 大纲 ----------
    private void OutlineList_ItemClick(object s, ItemClickEventArgs e)
    {
        if (e.ClickedItem is TextBlock tb && tb.Tag is int line)
            Post(new { type = BridgeMessages.ScrollToHeading, line });
        OutlinePopup.IsOpen = false;
    }

    private void ShowOutline()
    {
        OutlineList.Items.Clear();
        foreach (var (level, text, line) in _outline)
        {
            var tb = new TextBlock { Text = text, Tag = line, TextTrimming = TextTrimming.CharacterEllipsis };
            tb.Margin = new Thickness((level - 1) * 14, 4, 0, 4);
            OutlineList.Items.Add(tb);
        }
        OutlinePopup.IsOpen = true;
    }
}
```
